{{- /* Schoont RawContent op en herschrijft interne links naar hun .md-variant. Aanroepen met de oorsprongpagina als context (niet met een string): .GetPage resolvet relatieve paden t.o.v. die pagina. Wie de tabellen ongemoeid wil laten, roept aan met (dict "page" . "tabellen" true). */ -}} {{- $p := . -}} {{- $tabellen := false -}} {{- if reflect.IsMap . -}} {{- $p = .page -}} {{- $tabellen = .tabellen -}} {{- end -}} {{- $content := $p.RawContent -}} {{- /* GFM kent geen tabel zonder koprij. In de gepubliceerde .md wordt table-without-header daarom een lijst; pandoc krijgt een lege koprij, want zonder koprij ziet die geen tabel. Aan die lege koprij herkent scripts/downloads/render-downloads.js de tabel weer. Moet vóór het strippen van de shortcodetags. */ -}} {{- range findRE `\{\{<\s*table-without-header\s*>\}\}[\s\S]*?\{\{<\s*/\s*table-without-header\s*>\}\}` $content -}} {{- $rijen := slice -}} {{- $koprij := "" -}} {{- range split . "\n" -}} {{- $regel := trim . " " -}} {{- if hasPrefix $regel "|" -}} {{- if findRE `^\|[\s:|-]+\|$` $regel -}} {{- $koprij = index (last 1 $rijen) 0 -}} {{- else -}} {{- $rijen = $rijen | append $regel -}} {{- end -}} {{- end -}} {{- end -}} {{- if $koprij }}{{ $rijen = after 1 $rijen }}{{ end -}} {{- $regels := slice -}} {{- if $tabellen -}} {{- $leeg := slice -}} {{- $streep := slice -}} {{- range seq (len (split (trim (index $rijen 0) "|") "|")) -}} {{- $leeg = $leeg | append " " -}} {{- $streep = $streep | append " --- " -}} {{- end -}} {{- $regels = $regels | append (printf "|%s|" (delimit $leeg "|")) (printf "|%s|" (delimit $streep "|")) -}} {{- range $rijen }}{{ $regels = $regels | append . }}{{ end -}} {{- else -}} {{- range $rijen -}} {{- $cellen := apply (split (trim . "|") "|") "trim" "." " " -}} {{- $regels = $regels | append (printf "- **%s:** %s" (index $cellen 0) (delimit (after 1 $cellen) ", ")) -}} {{- end -}} {{- end -}} {{- $content = replace $content . (delimit $regels "\n") -}} {{- end -}} {{- $content = replaceRE `\{\{<[^>]+>\}\}` "" $content -}} {{- $content = replaceRE `\{\{%[^%]+%\}\}` "" $content -}} {{- $content = replaceRE `>\s*\[!(NOTE|WARNING|TIP|IMPORTANT|INFO|CAUTION)\]` "" $content -}} {{- /* findRESubmatch-groepen: 0=geheel, 1='!', 2=tekst, 3=doel */ -}} {{- range findRESubmatch `(!?)\[([^\]]*)\]\(([^)\s]+)\)` $content -}} {{- $whole := index . 0 -}} {{- $text := index . 2 -}} {{- $dest := index . 3 -}} {{- if eq (index . 1) "!" }}{{ continue }}{{ end -}} {{- if hasPrefix $dest "<" }}{{ continue }}{{ end -}} {{- $u := urls.Parse $dest -}} {{- if or $u.IsAbs (hasPrefix $dest "#") (hasPrefix $dest "//") (hasPrefix $dest "mailto:") (hasPrefix $dest "tel:") (not $u.Path) }}{{ continue }}{{ end -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $new := "" -}} {{- $target := $p.GetPage $path -}} {{- with $target -}} {{- $isMd := false -}} {{- with .File }}{{ if eq .Ext "md" }}{{ $isMd = true }}{{ end }}{{ end -}} {{- $mdOut := .OutputFormats.Get "markdown" -}} {{- if and $isMd $mdOut -}} {{- $new = $mdOut.RelPermalink -}} {{- else -}} {{- $new = .RelPermalink -}} {{- end -}} {{- else -}} {{- with or ($p.Resources.Get $path) (resources.Get $path) -}} {{- $new = .RelPermalink -}} {{- end -}} {{- end -}} {{- if $new -}} {{- with $u.RawQuery }}{{ $new = printf "%s?%s" $new . }}{{ end -}} {{- with $u.Fragment }}{{ $new = printf "%s#%s" $new . }}{{ end -}} {{- $content = replace $content $whole (printf "[%s](%s)" $text $new) -}} {{- end -}} {{- end -}} {{- $content = replaceRE `\n{3,}` "\n\n" $content -}} {{- return ($content | strings.TrimSpace | safeHTML) -}}
